## 1. 题目描述

- [leetcode](https://leetcode.cn/problems/intersection-of-two-arrays-ii/)

给你两个整数数组 `nums1` 和 `nums2`，请你以数组形式返回两数组的交集。返回结果中每个元素出现的次数，应与元素在两个数组中都出现的次数一致（如果出现次数不一致，则考虑取较小值）。可以不考虑输出结果的顺序。

---

示例 1：

```txt
输入：nums1 = [1, 2, 2, 1], nums2 = [2, 2]
输出：[2, 2]
```

---

示例 2：

```txt
输入：nums1 = [4, 9, 5], nums2 = [9, 4, 9, 8, 4]
输出：[4, 9]
```

---

提示：

- `1 <= nums1.length, nums2.length <= 1000`
- `0 <= nums1[i], nums2[i] <= 1000`

---

进阶：

- 如果给定的数组已经排好序呢？你将如何优化你的算法？
- 如果 `nums1` 的大小比 `nums2` 小，哪种方法更优？
- 如果 `nums2` 的元素存储在磁盘上，内存是有限的，并且你不能一次加载所有的元素到内存中，你该怎么办？

## 2. s.1 - 哈希表统计法

::: code-group

```c [c]
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */

// C 没有内置哈希表，用开放寻址记录出现次数
static int hashPos(int x, int cap) {
    unsigned int h = (unsigned int)x;
    h ^= h >> 16;
    h *= 0x7feb352du;
    h ^= h >> 15;
    return (int)(h & (unsigned int)(cap - 1));
}

static int tableCap(int n) {
    int cap = 1;
    while (cap < n * 2) cap <<= 1;
    return cap;
}

// 找到 x 所在槽；create 为 1 时不存在就新建并返回，否则返回 -1
static int findSlot(int* keys, unsigned char* used, int cap, int x, int create) {
    int pos = hashPos(x, cap);
    while (used[pos]) {
        if (keys[pos] == x) return pos;
        pos = (pos + 1) & (cap - 1);
    }
    if (!create) return -1;
    used[pos] = 1;
    keys[pos] = x;
    return pos;
}

int* intersect(int* nums1, int nums1Size, int* nums2, int nums2Size, int* returnSize) {
    // 优化：让较小的数组构建哈希表
    if (nums1Size > nums2Size) {
        return intersect(nums2, nums2Size, nums1, nums1Size, returnSize);
    }

    int cap = tableCap(nums1Size);
    int* keys = (int*)malloc((size_t)cap * sizeof(int));
    int* counts = (int*)calloc((size_t)cap, sizeof(int));
    unsigned char* used = (unsigned char*)calloc((size_t)cap, 1);

    // 使用哈希表统计 nums1 中每个元素的出现次数
    for (int i = 0; i < nums1Size; i++) {
        int pos = findSlot(keys, used, cap, nums1[i], 1);
        counts[pos]++;
    }

    int* result = (int*)malloc((size_t)(nums1Size > 0 ? nums1Size : 1) * sizeof(int));
    *returnSize = 0;

    // 遍历 nums2，查找交集
    for (int i = 0; i < nums2Size; i++) {
        int pos = findSlot(keys, used, cap, nums2[i], 0);
        if (pos >= 0 && counts[pos] > 0) {
            result[(*returnSize)++] = nums2[i];
            counts[pos]--; // 减少计数
        }
    }

    free(keys);
    free(counts);
    free(used);
    return result;
}
```

```js [js]
/**
 * @param {number[]} nums1
 * @param {number[]} nums2
 * @return {number[]}
 */
var intersect = function (nums1, nums2) {
  // 优化：让较小的数组构建哈希表
  if (nums1.length > nums2.length) {
    return intersect(nums2, nums1)
  }

  // 使用哈希表统计 nums1 中每个元素的出现次数
  const map = new Map()
  for (const num of nums1) {
    map.set(num, (map.get(num) || 0) + 1)
  }

  const result = []

  // 遍历 nums2，查找交集
  for (const num of nums2) {
    if (map.has(num) && map.get(num) > 0) {
      result.push(num)
      map.set(num, map.get(num) - 1) // 减少计数
    }
  }

  return result
}
```

```py [py]
class Solution:
    def intersect(self, nums1: list[int], nums2: list[int]) -> list[int]:
        # 优化：让较小的数组构建哈希表
        if len(nums1) > len(nums2):
            return self.intersect(nums2, nums1)

        # 使用哈希表统计 nums1 中每个元素的出现次数
        count = {}
        for num in nums1:
            count[num] = count.get(num, 0) + 1

        result = []

        # 遍历 nums2，查找交集
        for num in nums2:
            if count.get(num, 0) > 0:
                result.append(num)
                count[num] -= 1  # 减少计数

        return result
```

:::

- 时间复杂度：$O(m + n)$，其中 m 和 n 分别是两个数组的长度。哈希表的插入和查找期望是 $O(1)$，两个数组各扫一遍
- 空间复杂度：$O(\min(m, n))$，哈希表只统计较短数组里每个值的出现次数，不计输出数组

算法思路：

- 用较短的数组建哈希表，记下每个值出现的次数
- 再扫另一个数组：表里这个值还有剩余次数，就把它写入结果，并把次数减一
- 每个值写入的次数等于它在两个数组里出现次数的较小值；输出顺序没有要求

## 3. s.2 - 排序 + 双指针

::: code-group

```c [c]
static int cmp(const void* a, const void* b) {
    int x = *(const int*)a;
    int y = *(const int*)b;
    return (x > y) - (x < y);
}

/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* intersect(int* nums1, int nums1Size, int* nums2, int nums2Size, int* returnSize) {
    // 先对两个数组排序
    qsort(nums1, nums1Size, sizeof(int), cmp);
    qsort(nums2, nums2Size, sizeof(int), cmp);

    int cap = nums1Size < nums2Size ? nums1Size : nums2Size;
    int* result = (int*)malloc((size_t)cap * sizeof(int));
    *returnSize = 0;

    int i = 0, j = 0;
    // 使用双指针查找交集
    while (i < nums1Size && j < nums2Size) {
        if (nums1[i] < nums2[j]) {
            i++;
        } else if (nums1[i] > nums2[j]) {
            j++;
        } else {
            // 找到交集元素
            result[(*returnSize)++] = nums1[i];
            i++;
            j++;
        }
    }

    return result;
}
```

```js [js]
/**
 * @param {number[]} nums1
 * @param {number[]} nums2
 * @return {number[]}
 */
var intersect = function (nums1, nums2) {
  // 先对两个数组排序
  nums1.sort((a, b) => a - b)
  nums2.sort((a, b) => a - b)

  const result = []
  let i = 0,
    j = 0

  // 使用双指针查找交集
  while (i < nums1.length && j < nums2.length) {
    if (nums1[i] < nums2[j]) {
      i++
    } else if (nums1[i] > nums2[j]) {
      j++
    } else {
      // 找到交集元素
      result.push(nums1[i])
      i++
      j++
    }
  }

  return result
}
```

```py [py]
class Solution:
    def intersect(self, nums1: list[int], nums2: list[int]) -> list[int]:
        # 先对两个数组排序
        nums1.sort()
        nums2.sort()

        result = []
        i = j = 0

        # 使用双指针查找交集
        while i < len(nums1) and j < len(nums2):
            if nums1[i] < nums2[j]:
                i += 1
            elif nums1[i] > nums2[j]:
                j += 1
            else:
                # 找到交集元素
                result.append(nums1[i])
                i += 1
                j += 1

        return result
```

:::

- 时间复杂度：$O(m \log m + n \log n)$，排序主导；排好之后双指针各走一遍，是 $O(m + n)$
- 空间复杂度：$O(\log m + \log n)$，排序的递归栈，不计输出数组

算法思路：

- 先把两个数组升序排序，相同的值会连在一起
- 双指针从左向右：较小的那边前进；相等就把这个值写入结果，两边一起前进
- 相等时只消耗一次，同一个值会按两边次数的较小值重复写入。数组若已经有序，可以省掉排序，时间降到 $O(m + n)$，额外空间是 $O(1)$

## 4. 【AI】回复「进阶」部分的 3 个问题

### 4.1. 数组已经排好序

原文：「如果给定的数组已经排好序呢？你将如何优化你的算法？」

两个数组都有序时，用 s.2 的双指针，把排序去掉。

- 时间从 $O(m \log m + n \log n)$ 降到 $O(m + n)$
- 额外空间是 $O(1)$，不计输出数组
- 不必再建哈希表。有序之后线性合并已经最优，哈希表还要 $O(\min(m, n))$ 的额外空间

较小的那边前进；相等就把这个值写入结果，两边一起前进。同一个值按两边次数的较小值重复写入。

### 4.2. `nums1` 比 `nums2` 小很多

原文：「如果 `nums1` 的大小比 `nums2` 小，哪种方法更优？」

用 s.1，并且只用较短的 `nums1` 建哈希表。

- 时间仍是 $O(m + n)$，大数组只扫一遍
- 空间是 $O(m)$，也就是 $O(\min(m, n))$
- s.2 要给大数组排序，时间是 $O(n \log n)$。$n \gg m$ 时比线性扫描贵

s.1 就是这个做法：哈希表建在短的那边，再逐个消耗次数。

### 4.3. `nums2` 在磁盘上，一次读不进内存

原文：「如果 `nums2` 的元素存储在磁盘上，内存是有限的，并且你不能一次加载所有的元素到内存中，你该怎么办？」

**`nums1` 放得进内存**

沿用 s.1。先把 `nums1` 的出现次数放进哈希表，再按块把 `nums2` 读进来：块里的值如果还有剩余次数，就写入结果并把次数减一。`nums2` 只顺序读一遍，适合磁盘。不要在磁盘上对 `nums2` 做普通排序，也不要对它二分：随机读比顺序扫一遍贵。

**两个数组都放不进内存**

对两边做外部排序（多路归并），排好后仍用 s.2 的双指针，顺序读出相等的值。归并和双指针都是顺序读写。

也可以把 `nums1` 切成能放进内存的块，每块建哈希表再整表扫一遍 `nums2`。结果是对的，但 `nums2` 会被反复读，块越多越慢，一般不如外部排序后各扫一遍。
