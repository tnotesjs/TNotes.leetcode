## 1. 题目描述

- [leetcode](https://leetcode.cn/problems/intersection-of-two-arrays/)

给定两个数组 `nums1` 和 `nums2`，返回它们的交集。输出结果中的每个元素一定是唯一的。我们可以不考虑输出结果的顺序。

---

示例 1：

```txt
输入：nums1 = [1, 2, 2, 1], nums2 = [2, 2]
输出：[2]
```

---

示例 2：

```txt
输入：nums1 = [4, 9, 5], nums2 = [9, 4, 9, 8, 4]
输出：[9, 4]
```

解释：`[4, 9]` 也是可通过的

---

提示：

- `1 <= nums1.length, nums2.length <= 1000`
- `0 <= nums1[i], nums2[i] <= 1000`

## 2. s.1 - 哈希集合法

::: code-group

```c [c]
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */

// C 没有内置集合，用开放寻址哈希表。值域有界时的定长数组见 s.3
static int setPos(int x, int cap) {
    unsigned int h = (unsigned int)x;
    h ^= h >> 16;
    h *= 0x7feb352du;
    h ^= h >> 15;
    return (int)(h & (unsigned int)(cap - 1));
}

// 插入成功返回 1，已存在返回 0
static int setAdd(int* keys, unsigned char* used, int cap, int x) {
    int pos = setPos(x, cap);
    while (used[pos]) {
        if (keys[pos] == x) return 0;
        pos = (pos + 1) & (cap - 1);
    }
    used[pos] = 1;
    keys[pos] = x;
    return 1;
}

static int setHas(int* keys, unsigned char* used, int cap, int x) {
    int pos = setPos(x, cap);
    while (used[pos]) {
        if (keys[pos] == x) return 1;
        pos = (pos + 1) & (cap - 1);
    }
    return 0;
}

static int tableCap(int n) {
    int cap = 1;
    while (cap < n * 2) cap <<= 1;
    return cap;
}

int* intersection(int* nums1, int nums1Size, int* nums2, int nums2Size, int* returnSize) {
    int cap1 = tableCap(nums1Size);
    int cap2 = tableCap(nums2Size);

    int* keys1 = (int*)malloc((size_t)cap1 * sizeof(int));
    int* keys2 = (int*)malloc((size_t)cap2 * sizeof(int));
    unsigned char* used1 = (unsigned char*)calloc((size_t)cap1, 1);
    unsigned char* used2 = (unsigned char*)calloc((size_t)cap2, 1);

    // 使用哈希表对两个数组去重
    int count1 = 0, count2 = 0;
    for (int i = 0; i < nums1Size; i++) {
        count1 += setAdd(keys1, used1, cap1, nums1[i]);
    }
    for (int i = 0; i < nums2Size; i++) {
        count2 += setAdd(keys2, used2, cap2, nums2[i]);
    }

    // 遍历较小的集合，提高效率
    int* smallKeys = count1 < count2 ? keys1 : keys2;
    unsigned char* smallUsed = count1 < count2 ? used1 : used2;
    int smallCap = count1 < count2 ? cap1 : cap2;
    int smallCount = count1 < count2 ? count1 : count2;
    int* largeKeys = count1 < count2 ? keys2 : keys1;
    unsigned char* largeUsed = count1 < count2 ? used2 : used1;
    int largeCap = count1 < count2 ? cap2 : cap1;

    int* result = (int*)malloc((size_t)(smallCount > 0 ? smallCount : 1) * sizeof(int));
    *returnSize = 0;

    // 查找交集元素
    for (int i = 0; i < smallCap; i++) {
        if (smallUsed[i] && setHas(largeKeys, largeUsed, largeCap, smallKeys[i])) {
            result[(*returnSize)++] = smallKeys[i];
        }
    }

    free(keys1);
    free(keys2);
    free(used1);
    free(used2);
    return result;
}
```

```js [js]
/**
 * @param {number[]} nums1
 * @param {number[]} nums2
 * @return {number[]}
 */
var intersection = function (nums1, nums2) {
  // 使用 Set 对两个数组去重
  const set1 = new Set(nums1)
  const set2 = new Set(nums2)

  // 存储交集结果
  const result = []

  // 遍历较小的集合，提高效率
  const [smallerSet, largerSet] =
    set1.size < set2.size ? [set1, set2] : [set2, set1]

  // 查找交集元素
  for (const num of smallerSet) {
    if (largerSet.has(num)) {
      result.push(num)
    }
  }

  return result
}
```

```py [py]
class Solution:
    def intersection(self, nums1: list[int], nums2: list[int]) -> list[int]:
        # 使用 set 对两个数组去重
        set1 = set(nums1)
        set2 = set(nums2)

        # 遍历较小的集合，提高效率
        if len(set1) > len(set2):
            set1, set2 = set2, set1

        # 查找交集元素
        result = []
        for num in set1:
            if num in set2:
                result.append(num)
        return result
```

:::

- 时间复杂度：$O(m + n)$，其中 m 和 n 分别是两个数组的长度。集合的插入和查找期望是 $O(1)$，两个数组各扫一遍
- 空间复杂度：$O(m + n)$，两个集合最多存下全部不重复元素

算法思路：

- 分别把两个数组放进集合，重复值只保留一份，交集里的元素天然唯一
- 只遍历元素更少的那个集合，能在另一个集合里查到的值就是交集
- 输出顺序没有要求，查到就写入结果

## 3. s.2 - 排序 + 双指针

::: code-group

```c [c]
static int cmp(const void* a, const void* b) {
    int x = *(const int*)a;
    int y = *(const int*)b;
    return (x > y) - (x < y);
}

/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* intersection(int* nums1, int nums1Size, int* nums2, int nums2Size, int* returnSize) {
    // 先对两个数组排序
    qsort(nums1, nums1Size, sizeof(int), cmp);
    qsort(nums2, nums2Size, sizeof(int), cmp);

    int cap = nums1Size < nums2Size ? nums1Size : nums2Size;
    int* result = (int*)malloc((size_t)cap * sizeof(int));
    *returnSize = 0;

    int i = 0, j = 0;
    // 使用双指针查找交集
    while (i < nums1Size && j < nums2Size) {
        if (nums1[i] < nums2[j]) {
            i++;
        } else if (nums1[i] > nums2[j]) {
            j++;
        } else {
            // 找到交集元素，注意：避免重复添加。
            if (*returnSize == 0 || result[*returnSize - 1] != nums1[i]) {
                result[(*returnSize)++] = nums1[i];
            }
            i++;
            j++;
        }
    }

    return result;
}
```

/**

 * @param {number[]} nums2
 * @return {number[]}

var intersection = function (nums1, nums2) {
  // 先对两个数组排序
  nums1.sort((a, b) => a - b)
  nums2.sort((a, b) => a - b)

  const result = []
  let i = 0,
    j = 0

  // 使用双指针查找交集
  while (i < nums1.length && j < nums2.length) {
    if (nums1[i] < nums2[j]) {
      i++
    } else if (nums1[i] > nums2[j]) {
      j++
    } else {
      // 找到交集元素，注意：避免重复添加。
      if (result.length === 0 || result[result.length - 1] !== nums1[i]) {
        result.push(nums1[i])
      }
      i++
      j++
    }
  }

  return result
}

```py [py]
class Solution:
    def intersection(self, nums1: list[int], nums2: list[int]) -> list[int]:
        # 先对两个数组排序
        nums1.sort()
        nums2.sort()

        result = []
        i = j = 0

        # 使用双指针查找交集
        while i < len(nums1) and j < len(nums2):
            if nums1[i] < nums2[j]:
                i += 1
            elif nums1[i] > nums2[j]:
                j += 1
            else:
                # 找到交集元素，注意：避免重复添加。
                if not result or result[-1] != nums1[i]:
                    result.append(nums1[i])
                i += 1
                j += 1

        return result
```

:::

- 时间复杂度：$O(m \log m + n \log n)$，排序主导；排好之后双指针各走一遍，是 $O(m + n)$
- 空间复杂度：$O(\log m + \log n)$，排序的递归栈，不计输出数组

算法思路：

- 先把两个数组升序排序，相同的值会连在一起
- 双指针从左向右：较小的那边前进；相等就记录这个值，两边一起前进
- 记录前看结果末尾是不是已经是这个值，避免重复加入

## 4. s.3 - 布尔数组（桶）

::: code-group

```c [c]
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* intersection(int* nums1, int nums1Size, int* nums2, int nums2Size, int* returnSize) {
    int seen[1001] = {0};
    int cap = nums1Size < nums2Size ? nums1Size : nums2Size;
    int* result = (int*)malloc((size_t)cap * sizeof(int));
    *returnSize = 0;

    for (int i = 0; i < nums1Size; i++) {
        seen[nums1[i]] = 1;
    }
    for (int i = 0; i < nums2Size; i++) {
        if (seen[nums2[i]]) {
            result[(*returnSize)++] = nums2[i];
            seen[nums2[i]] = 0; // 避免重复添加
        }
    }
    return result;
}
```

```js [js]
/**
 * @param {number[]} nums1
 * @param {number[]} nums2
 * @return {number[]}
 */
var intersection = function(nums1, nums2) {
  const seen = new Array(1001).fill(false);
  const result = [];
  for (const num of nums1) {
    seen[num] = true;
  }
  for (const num of nums2) {
    if (seen[num]) {
      result.push(num);
      seen[num] = false; // 避免重复添加
    }
  }
  return result;
}
```

```py [py]
class Solution:
    def intersection(self, nums1: list[int], nums2: list[int]) -> list[int]:
        seen = [False] * 1001
        result = []
        for num in nums1:
            seen[num] = True
        for num in nums2:
            if seen[num]:
                result.append(num)
                seen[num] = False  # 避免重复添加
        return result
```

:::

- 时间复杂度：$O(m + n)$，两个数组各扫一遍
- 空间复杂度：$O(1)$，桶的长度由值域上界 $1000$ 固定为 $1001$，不计输出数组

算法思路：

- 每个数都在 $[0, 1000]$，开一个长度 $1001$ 的布尔数组当桶
- 先扫 `nums1`，出现过的值标成真
- 再扫 `nums2`，桶里为真的就是交集；写入后立刻改回假，同一个值不会重复加入
