## 1. 📝 题目描述

- [leetcode](https://leetcode.cn/problems/make-array-empty/)

给你一个包含若干 互不相同 整数的数组 `nums`，你需要执行以下操作 直到数组为空 ：

- 如果数组中第一个元素是当前数组中的 最小值，则删除它。
- 否则，将第一个元素移动到数组的 末尾。

请你返回需要多少个操作使 `nums` 为空。

---

示例 1：

```txt
输入：nums = [3,4,-1]
输出：5
```

| Operation | Array        |
| --------- | ------------ |
| `1`       | `[4, -1, 3]` |
| `2`       | `[-1, 3, 4]` |
| `3`       | `[3, 4]`     |
| `4`       | `[4]`        |
| `5`       | `[]`         |

---

示例 2：

```txt
输入：nums = [1,2,4,3]
输出：5
```

| Operation | Array       |
| --------- | ----------- |
| `1`       | `[2, 4, 3]` |
| `2`       | `[4, 3]`    |
| `3`       | `[3, 4]`    |
| `4`       | `[4]`       |
| `5`       | `[]`        |

---

示例 3：

```txt
输入：nums = [1,2,3]
输出：3
```

| Operation | Array    |
| --------- | -------- |
| `1`       | `[2, 3]` |
| `2`       | `[3]`    |
| `3`       | `[]`     |

---

提示：

- `1 <= nums.length <= 10^5`
- `-10^9 <= nums[i] <= 10^9`
- `nums` 中的元素 互不相同。

```js [js]
/**
 * @param {number[]} nums
 * @return {number}
 */
var countOperationsToEmptyArray = function (nums) {
  const n = nums.length
  const indices = Array.from({ length: n }, (_, i) => i)
  indices.sort((a, b) => nums[a] - nums[b])

  // BIT
  const bit = new Array(n + 2).fill(0)
  function update(i, val) {
    for (i += 1; i <= n; i += i & -i) bit[i] += val
  }
  function query(i) {
    let s = 0
    for (i += 1; i > 0; i -= i & -i) s += bit[i]
    return s
  }
  function rangeQuery(l, r) {
    if (l > r) return 0
    return query(r) - (l > 0 ? query(l - 1) : 0)
  }

  for (let i = 0; i < n; i++) update(i, 1)

  let ops = n
  let prev = -1
  for (let k = 0; k < n; k++) {
    const cur = indices[k]
    if (prev === -1) {
      ops += cur
    } else if (cur > prev) {
      ops += rangeQuery(prev + 1, cur - 1)
    } else {
      ops += rangeQuery(prev + 1, n - 1) + rangeQuery(0, cur - 1)
    }
    update(cur, -1)
    prev = cur
  }
  return ops
}
```

```c [c]
            if (prev + 1 <= n - 1) cnt1 = query(n - 1) - query(prev);
            if (cur - 1 >= 0) cnt2 = query(cur - 1);
            ops += cnt1 + cnt2;
        }
        update(cur, -1);
        prev = cur;
    }
    free(indices);
    free(bit);
    return ops;
}
```

```py [py]
class Solution:
    def countOperationsToEmptyArray(self, nums: list[int]) -> int:
        n = len(nums)
        indices = sorted(range(n), key=lambda i: nums[i])

        bit = [0] * (n + 2)
        def update(i: int, val: int) -> None:
            i += 1
            while i <= n:
                bit[i] += val
                i += i & (-i)
        def query(i: int) -> int:
            s = 0
            i += 1
            while i > 0:
                s += bit[i]
                i -= i & (-i)
            return s
        def range_query(l: int, r: int) -> int:
            if l > r:
                return 0
            return query(r) - (query(l - 1) if l > 0 else 0)

        for i in range(n):
            update(i, 1)

        ops = n
        prev = -1
        for cur in indices:
            if prev == -1:
                ops += cur
            elif cur > prev:
                ops += range_query(prev + 1, cur - 1)
            else:
                ops += range_query(prev + 1, n - 1) + range_query(0, cur - 1)
            update(cur, -1)
            prev = cur
        return ops
```

:::

- 时间复杂度：$O(N \log N)$，排序和树状数组操作
- 空间复杂度：$O(N)$，存储排序后的索引和树状数组

算法思路：

- 按元素值对原数组索引排序，依次按从小到大顺序删除元素
- 总操作数 = N（每个元素删除一次）+ 所有「移到末尾」操作的总次数
- 用树状数组维护每个位置是否存活，快速计算从上一个删除位置到当前删除位置之间（环形）还有多少存活元素需要移到末尾
